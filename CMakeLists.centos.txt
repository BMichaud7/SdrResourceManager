cmake_minimum_required(VERSION 3.20)
project(sdr_controller VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ── Compiler flags ───────────────────────────────────────────────────────────
add_compile_options(
    -Wall -Wextra -Wpedantic
    -Wno-unused-parameter
    $<$<CONFIG:Release>:-O3>
    $<$<CONFIG:Debug>:-g>
    $<$<CONFIG:Debug>:-fsanitize=address>
    $<$<CONFIG:Debug>:-fsanitize=undefined>
)
add_link_options(
    $<$<CONFIG:Debug>:-fsanitize=address>
    $<$<CONFIG:Debug>:-fsanitize=undefined>
)

# ── PkgConfig ────────────────────────────────────────────────────────────────
find_package(PkgConfig REQUIRED)

# ── SoapySDR ─────────────────────────────────────────────────────────────────
find_package(SoapySDR REQUIRED)

# ── qpid-proton C++ (AMQP 1.0) ───────────────────────────────────────────────
# On RHEL/CentOS the pkg-config name is "libqpid-proton-cpp" (with lib prefix).
# Ubuntu uses "qpid-proton-cpp". Confirmed present via diagnostic output.
pkg_check_modules(PROTON REQUIRED libqpid-proton-cpp)

# ── TinyXML2 ─────────────────────────────────────────────────────────────────
pkg_check_modules(TINYXML2 REQUIRED tinyxml2)

# ── spdlog (built from source into /usr/local) ───────────────────────────────
find_package(spdlog REQUIRED
    PATHS /usr/local/lib64/cmake/spdlog
    NO_DEFAULT_PATH)

# ── fmt (system package, needed by spdlog EXTERNAL build) ────────────────────
find_package(fmt REQUIRED
    PATHS /usr/lib64/cmake/fmt
    NO_DEFAULT_PATH)

# ── nlohmann/json ─────────────────────────────────────────────────────────────
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
    include(FetchContent)
    FetchContent_Declare(
        nlohmann_json
        GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG        v3.11.3
    )
    FetchContent_MakeAvailable(nlohmann_json)
endif()

# ── FFTW3 single-precision ────────────────────────────────────────────────────
find_library(FFTW3F_LIB NAMES fftw3f REQUIRED)
find_path(FFTW3_INCLUDE NAMES fftw3.h REQUIRED)

# ── libuuid ───────────────────────────────────────────────────────────────────
find_library(UUID_LIB NAMES uuid REQUIRED)
find_path(UUID_INCLUDE NAMES uuid/uuid.h REQUIRED)

# ── Threads ───────────────────────────────────────────────────────────────────
find_package(Threads REQUIRED)

# ── Controller executable ─────────────────────────────────────────────────────
file(GLOB CONTROLLER_SOURCES src/*.cpp)
add_executable(sdr_controller ${CONTROLLER_SOURCES})

target_include_directories(sdr_controller PRIVATE
    include
    ${SoapySDR_INCLUDE_DIRS}
    ${PROTON_INCLUDE_DIRS}
    ${TINYXML2_INCLUDE_DIRS}
    ${FFTW3_INCLUDE}
    ${UUID_INCLUDE}
)

target_link_libraries(sdr_controller PRIVATE
    SoapySDR
    ${PROTON_LIBRARIES}
    ${TINYXML2_LIBRARIES}
    spdlog::spdlog
    fmt::fmt
    ${FFTW3F_LIB}
    ${UUID_LIB}
    Threads::Threads
    nlohmann_json::nlohmann_json
)

target_compile_definitions(sdr_controller PRIVATE
    SPDLOG_ACTIVE_LEVEL=SPDLOG_LEVEL_DEBUG
    SPDLOG_FMT_EXTERNAL
)

# ── Client ────────────────────────────────────────────────────────────────────
add_subdirectory(client)

# ── Tests ─────────────────────────────────────────────────────────────────────
enable_testing()
add_subdirectory(tests)

# ── Install ───────────────────────────────────────────────────────────────────
install(TARGETS sdr_controller DESTINATION bin)
install(FILES config/devices.xml DESTINATION etc/sdr-controller)
