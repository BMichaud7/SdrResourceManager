cmake_minimum_required(VERSION 3.20)
find_package(PkgConfig REQUIRED)

# RHEL/CentOS pkg-config name is "libqpid-proton-cpp" (lib prefix)
pkg_check_modules(PROTON REQUIRED libqpid-proton-cpp)

find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
    include(FetchContent)
    FetchContent_Declare(
        nlohmann_json
        GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG        v3.11.3
    )
    FetchContent_MakeAvailable(nlohmann_json)
endif()

find_library(UUID_LIB NAMES uuid REQUIRED)
find_path(UUID_INCLUDE NAMES uuid/uuid.h REQUIRED)
find_package(Threads REQUIRED)

add_executable(sdr_client sdr_client.cpp)

target_include_directories(sdr_client PRIVATE
    ${PROTON_INCLUDE_DIRS}
    ${UUID_INCLUDE}
    ${CMAKE_SOURCE_DIR}/include
)

target_link_libraries(sdr_client PRIVATE
    ${PROTON_LIBRARIES}
    ${UUID_LIB}
    Threads::Threads
    nlohmann_json::nlohmann_json
)

target_compile_features(sdr_client PRIVATE cxx_std_20)
install(TARGETS sdr_client DESTINATION bin)
